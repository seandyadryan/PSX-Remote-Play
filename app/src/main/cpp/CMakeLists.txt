cmake_minimum_required(VERSION 3.22.1)
project(psx_connection LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Media/FEC code needs optimization even in the debuggable app build.
add_compile_options(-O2)
set(DEPS "${CMAKE_CURRENT_LIST_DIR}/../../../build/native-provider")
set(CHIAKI "${DEPS}/chiaki-ng-ca5379f4504af6e63cf5d32b5626de2425b2382e")
set(NANOPB "${DEPS}/nanopb-afc499f9a410fc9bbf6c9c48cdd8d8b199d49eb4")
set(GF "${DEPS}/gf-complete-a6862d10c9db467148f20eef2c6445ac9afd94d8")
set(JERASURE "${DEPS}/jerasure-de1739cc8483696506829b52e7fda4f6bb195e6a")
if(NOT EXISTS "${CHIAKI}/lib/src/session.c")
  message(FATAL_ERROR "Run scripts/bootstrap-native.ps1 before configuring native code")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/mbedtls-2.28.10" mbedtls EXCLUDE_FROM_ALL)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/opus-1.5.2" opus EXCLUDE_FROM_ALL)
add_library(nanopb STATIC "${NANOPB}/pb_common.c" "${NANOPB}/pb_decode.c" "${NANOPB}/pb_encode.c")
target_include_directories(nanopb PUBLIC "${NANOPB}")
file(GLOB GF_SOURCES CONFIGURE_DEPENDS "${GF}/src/gf*.c")
add_library(gf_complete STATIC ${GF_SOURCES})
target_include_directories(gf_complete PUBLIC "${GF}/include")
# This older portable C source calls bzero without including its POSIX declaration.
target_compile_options(gf_complete PRIVATE -include strings.h)
add_library(jerasure STATIC "${JERASURE}/src/galois.c" "${JERASURE}/src/jerasure.c" "${JERASURE}/src/reed_sol.c" "${JERASURE}/src/cauchy.c" "${JERASURE}/src/liberation.c")
target_include_directories(jerasure PUBLIC "${JERASURE}/include")
target_compile_options(jerasure PRIVATE -include strings.h)
target_link_libraries(jerasure PUBLIC gf_complete)
file(GLOB CHIAKI_SOURCES CONFIGURE_DEPENDS "${CHIAKI}/lib/src/*.c")
list(FILTER CHIAKI_SOURCES EXCLUDE REGEX "/(ffmpegdecoder|pidecoder)\\.c$")
set(CHIAKI_LIB_ENABLE_OPUS ON)
set(CHIAKI_LIB_ENABLE_PI_DECODER OFF)
configure_file("${CHIAKI}/lib/config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/include/chiaki/config.h")
add_library(chiaki-lib STATIC ${CHIAKI_SOURCES} generated/takion.pb.c)
target_include_directories(chiaki-lib PUBLIC "${CHIAKI}/lib/include" "${CMAKE_CURRENT_BINARY_DIR}/include" PRIVATE generated "${DEPS}/opus-1.5.2/include/..")
target_compile_definitions(chiaki-lib PUBLIC CHIAKI_LIB_ENABLE_MBEDTLS CHIAKI_VERSION_MAJOR=1 CHIAKI_VERSION_MINOR=6 CHIAKI_VERSION_PATCH=5 CHIAKI_VERSION="1.6.5")
# Upstream's headers use <opus/opus.h>; mirror the include prefix at configure time.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/opus")
file(GLOB OPUS_HEADERS "${DEPS}/opus-1.5.2/include/*.h")
file(COPY ${OPUS_HEADERS} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/opus")
target_link_libraries(chiaki-lib PUBLIC mbedtls mbedx509 mbedcrypto nanopb jerasure opus m)
add_library(psx-connection SHARED native_bridge.cpp)
target_link_libraries(psx-connection PRIVATE chiaki-lib android log)
target_link_options(psx-connection PRIVATE "-Wl,-z,max-page-size=16384")
